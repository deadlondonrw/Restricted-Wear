cmake_minimum_required(VERSION 3.22)

project(DEADNOISE VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(DEADNOISE_BUILD_PLUGIN "Build the JUCE VST3 plugin" ON)
option(DEADNOISE_BUILD_TESTS  "Build the native DSP tests and the offline CLI" ON)
set(DEADNOISE_JUCE_DIR "" CACHE PATH "Path to a JUCE 8 checkout (fetched automatically when empty)")

# ------------------------------------------------------------------------------
# Framework-independent DSP core (the actual noise suppression)
add_library(deadnoise_dsp STATIC
    src/dsp/SpectralDenoiser.cpp
    src/dsp/NoiseProfile.cpp
    src/dsp/DeadnoiseEngine.cpp)
target_include_directories(deadnoise_dsp PUBLIC src/dsp)
set_target_properties(deadnoise_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(deadnoise_dsp PRIVATE /W4 /O2 /fp:precise)
else()
    target_compile_options(deadnoise_dsp PRIVATE -Wall -Wextra -O3 -fno-math-errno)
endif()

# ------------------------------------------------------------------------------
if(DEADNOISE_BUILD_PLUGIN)
    if(DEADNOISE_JUCE_DIR)
        add_subdirectory(${DEADNOISE_JUCE_DIR} ${CMAKE_BINARY_DIR}/JUCE EXCLUDE_FROM_ALL)
    else()
        include(FetchContent)
        FetchContent_Declare(JUCE
            GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
            GIT_TAG 8.0.15
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(JUCE)
    endif()

    # moduleinfo.json is generated by running a helper on the build machine,
    # which is impossible when cross-compiling.
    if(CMAKE_CROSSCOMPILING)
        set(DEADNOISE_MANIFEST FALSE)
    else()
        set(DEADNOISE_MANIFEST TRUE)
    endif()

    juce_add_plugin(DEADNOISE
        PRODUCT_NAME "DEADNOISE"
        COMPANY_NAME "DEADNOISE"
        BUNDLE_ID "com.deadnoise.deadnoise"
        PLUGIN_MANUFACTURER_CODE Ddns
        PLUGIN_CODE Dnse
        FORMATS VST3
        IS_SYNTH FALSE
        NEEDS_MIDI_INPUT FALSE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        VST3_CATEGORIES Fx Restoration
        VST3_AUTO_MANIFEST ${DEADNOISE_MANIFEST}
        COPY_PLUGIN_AFTER_BUILD FALSE)

    target_sources(DEADNOISE PRIVATE
        src/plugin/PluginProcessor.cpp
        src/plugin/PluginEditor.cpp)

    target_compile_definitions(DEADNOISE PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_DISPLAY_SPLASH_SCREEN=0
        JUCE_REPORT_APP_USAGE=0
        JUCE_STRICT_REFCOUNTEDPOINTER=1
        DEADNOISE_VERSION_STRING="${PROJECT_VERSION}")

    target_link_libraries(DEADNOISE
        PRIVATE
            deadnoise_dsp
            juce::juce_audio_utils
            juce::juce_audio_processors
            juce::juce_gui_basics
        PUBLIC
            juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags)

    if(MINGW)
        # self-contained DLL: no libc++ / libunwind / winpthread DLLs needed
        target_link_options(DEADNOISE_VST3 PRIVATE -static -Wl,--gc-sections)
        target_link_options(DEADNOISE PRIVATE -static)
    endif()
endif()

# ------------------------------------------------------------------------------
# Host-level test (loads the built .vst3 like a DAW) and UI screenshot tool
if(DEADNOISE_BUILD_PLUGIN AND DEADNOISE_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    juce_add_console_app(deadnoise_host_test PRODUCT_NAME "deadnoise_host_test")
    target_sources(deadnoise_host_test PRIVATE tools/host_test.cpp)
    target_compile_definitions(deadnoise_host_test PRIVATE
        JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
    target_link_libraries(deadnoise_host_test PRIVATE
        deadnoise_dsp juce::juce_audio_processors juce::juce_gui_basics
        juce::juce_recommended_config_flags)

    juce_add_console_app(deadnoise_screenshot PRODUCT_NAME "deadnoise_screenshot")
    target_sources(deadnoise_screenshot PRIVATE
        tools/screenshot.cpp src/plugin/PluginProcessor.cpp src/plugin/PluginEditor.cpp)
    target_compile_definitions(deadnoise_screenshot PRIVATE
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 DEADNOISE_VERSION_STRING="${PROJECT_VERSION}")
    target_link_libraries(deadnoise_screenshot PRIVATE
        deadnoise_dsp juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_basics
        juce::juce_recommended_config_flags)
endif()

if(DEADNOISE_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    add_executable(deadnoise_tests tests/dsp_tests.cpp)
    target_link_libraries(deadnoise_tests PRIVATE deadnoise_dsp)
    target_include_directories(deadnoise_tests PRIVATE tools)

    add_executable(deadnoise_cli tools/deadnoise_cli.cpp)
    target_link_libraries(deadnoise_cli PRIVATE deadnoise_dsp)

    find_package(Threads REQUIRED)
    target_link_libraries(deadnoise_tests PRIVATE Threads::Threads)
    target_link_libraries(deadnoise_cli PRIVATE Threads::Threads)

    enable_testing()
    add_test(NAME dsp_tests COMMAND deadnoise_tests)
endif()
